 21: #ifndef _LISTALIN_H
 22: #define _LISTALIN_H

 24: #include <iostream>

 26: using std::cout;
 27: using std::endl;
 28: using std::ostream;

 30: // classe astratta listaLinare
 31: template< class T, class P >
 32: class Linear_list{
 33:  public:
 34:         typedef T value_type;   // the type of object, T, stored in the list
 35:         typedef P position;
 36:         
 37:         // operators
 38:         virtual void create() = 0;
 39:         virtual bool empty() const = 0; // true if the list's size is 0
 40:         virtual value_type read(position) const = 0;
 41:         virtual void write(const value_type & x, position p) = 0; // write x at position p
 42:         virtual position begin() const = 0;  // returns a position pointing to the beginning of the list
 43:         virtual bool end(position) const = 0; // returns a position pointing to the end of the list
 44:         virtual position next(position) const = 0; // returns the next position
 45:         virtual position previous(position) const = 0; // return the previous position
 46:         virtual void insert(const value_type &, position) = 0; // insert an element 
 47:         virtual void erase(position pos) = 0; // ersaes the element at position pos
 48:         
 49:         // funzioni di servizio
 50:         //friend ostream& operator<< <T,P>(ostream&, const Linear_list<T,P>&);

 52:         /* Altre funzioni di servizio implementabili */
 53:         /*
 54:                 virtual int size() const;  // returns the size of the list
 55:                 virtual void push_front(const value_type &); // insert a new element at the beginning
 56:                 virtual void push_back(const value_type &); // insert a new element at the end
 57:                 virtual void pop_front(); // removes the first element
 58:                 virtual void pop_back(); // removes the last element
 59:                 virtual void clear(); // erases all the elements
 60:         */

 62: };

 64: /* sovraccarica <<. Attenzione se il tipo restituito da read non Ã¨ primitivo, allora
 65:  * anche per questo tipo bisogna sovraccaricare l'operatore << 
 66:  */
 67: template< class T, class P >
 68: ostream& operator<<(ostream& os, const Linear_list<T,P> &l){
 69:         typename Linear_list<T,P>::position p;
 70:         p = l.begin();
 71:         cout << "[";
 72:         while (!l.end(p)){
 73:                 if (p != l.begin())
 74:                         cout << ", " << l.read(p);
 75:                 else
 76:                         cout << l.read(p);
 77:                 p = l.next(p);
 78:         }
 79:         cout << "]" << endl;
 80:         return os;
 81: }

 83: #endif // _LISTALIN_H
