 21: #ifndef _LISTAVT_H
 22: #define _LISTAVT_H

 24: #include "linear_list.h"

 26: // classe Lista
 27: template< class T >
 28: class List_vector : public Linear_list<T, int>{
 29:  public:
 30:         typedef typename Linear_list<T, int>::value_type value_type;
 31:         typedef typename Linear_list<T, int>::position position;

 33:         // costruttori
 34:         List_vector();    
 35:         List_vector(int); 
 36:         // costruttore per copia
 37:         List_vector(const List_vector<T>& ); 
 38:         //distruttore
 39:         ~List_vector();   
 40:         
 41:         // operatori
 42:         void create();
 43:         bool empty() const;
 44:         value_type read(position) const;
 45:         void write(const value_type &, position);
 46:         position begin() const;
 47:         bool end(position) const;
 48:         position next(position) const;
 49:         position previous(position) const;
 50:         void insert(const value_type &, position);
 51:         void erase(position);

 53:         // sovraccarico di operatori
 54:         List_vector<T>& operator=(const List_vector<T>&); // the assignment operator
 55:         bool operator==(const List_vector<T> &) const; // tests two list for equality

 57:  private:
 58:         void change_dimension_(T*& , int , int );
 59:         value_type* elements_;
 60:         int length_; // the length of the list
 61:         int array_dimension_; // array's dimension
 62: };

 64: // costruttore
 65: template< class T >
 66: List_vector< T >::List_vector() {
 67:   array_dimension_ = 10;
 68:   this->create();
 69: }

 71: // costruttore
 72: template< class T >
 73: List_vector< T >::List_vector(int dim){
 74:   array_dimension_ = dim;
 75:   this->create();
 76: }

 78: /* Il costruttore per copia effettua una copia o clone di un oggetto.  
 79:  * Questo costruttore viene invocato, per esempio, quando un oggetto viene passato 
 80:  * per valore ad una funzione o quando una funzione restituisce un oggetto. 
 81:  */
 82: template< class T >
 83: List_vector< T >::List_vector(const List_vector<T>& Lista) {
 84:   this->array_dimension_ = Lista.array_dimension_;
 85:   this->length_ = Lista.length_;
 86:   this->elements_ = new T[array_dimension_];
 87:   for (int i=0; i<Lista.array_dimension_; i++)
 88:      this->elements_[i] = Lista.elements_[i];
 89: }

 91: // distruttore
 92: template< class T >
 93: List_vector< T >::~List_vector(){
 94:         delete elements_;
 95: }

 97: // operatori
 98: template< class T >
 99: void List_vector< T >::create(){
100:   this->elements_ = new T[array_dimension_];
101:   this->length_ = 0;
102: }

104: template< class T >
105: bool List_vector< T >::empty() const {
106:     return(length_ == 0);
107: }

109: template< class T >
110: typename List_vector< T >::position List_vector< T >::begin() const {
111:         return(1); // e quindi pos(1)=pos(n+1) se la lista -Ah vuota (n=0)
112: }

114: template< class T >
115: typename List_vector< T >::position List_vector< T >::next(position p) const {
116:         if ( (0 < p) && (p < length_+1)) // precondizione
117:                 return(p+1);
118:         else
119:                 return(p);
120: }

122: template< class T >
123: typename List_vector< T >::position List_vector< T >::previous(position p) const {
124:         if ( (1 < p) && (p < length_+1)) // precondizione
125:                 return(p-1);
126:         else
127:                 return(p);
128: }

130: template< class T >
131: bool List_vector< T >::end(position p) const {
132:         if ( (0 < p) && (p <= length_+1)) // precondizione
133:                 return( p == length_+1);
134:         else
135:                 return(false);
136: }

138: template< class T >
139: typename List_vector< T >::value_type List_vector< T >::read(position p) const {
140:         if ( (0 < p) && (p < length_+1)) // precondizione
141:                 return(elements_[p-1]);
142: }

144: template< class T >
145: void List_vector< T >::write(const value_type &a, position p) {
146:   if ( (0 < p) && (p < length_+1)) // precondizione
147:     elements_[p-1] = a;
148: }

150: template< class T >
151: void List_vector< T >::insert(const value_type &a, position p){
152:   if (length_ == array_dimension_){
153:     change_dimension_(elements_, array_dimension_, array_dimension_ * 2);
154:     array_dimension_ = array_dimension_ * 2;
155:   }
156:   if ( (0 < p) && (p <= length_+1)) { // precondizione
157:                 for (int i=length_; i>=p; i--)
158:                         elements_[i] = elements_[i-1];
159:                 elements_[p-1]=a;
160:                 length_++;
161:         }
162: }

164: template< class T >
165: void List_vector< T >::erase(position p){
166:   if ( (0 < p) && ( p < length_ + 1)) // precondizione
167:     if (!empty()) {
168:                         for (int i=p-1;i<(length_-1);i++)
169:                                 elements_[i]=elements_[i+1];
170:                         length_--;
171:                 }
172: }

174: template<class T>
175: void List_vector< T >::change_dimension_(T*& a, int vecchiaDim, int nuovaDim){

177:   T* temp = new T[nuovaDim];
178:   int number;
179:   if (vecchiaDim < nuovaDim) 
180:      number = vecchiaDim;
181:   else
182:      number = nuovaDim;
183:   for (int i=0; i<number; i++)
184:      temp[i]=a[i];
185:   delete [] a;
186:   a = temp;
187: }

189: /* operatore di assegnamento */
190: template<class T>
191: List_vector<T>& List_vector<T>::operator=(const List_vector<T>& l){
192:         if (this != &l){   // attenzione agli autoassegnamenti: l = l 
193:                 this->array_dimension_ = l.array_dimension_;
194:                 this->length_ = l.length_;
195:                 delete this->elements_;
196:                 this->elements_ = new T[array_dimension_];
197:                 for (int i=0; i<l.array_dimension_; i++)
198:                         this->elements_[i] = l.elements_[i];
199:         }
200:         return *this;
201: }

203: /* operatore di test di uguaglianza */
204: template<class T>
205: bool List_vector<T>::operator==(const List_vector<T> &l) const{
206:         if (l.length_ != this->length_)
207:                 return false;
208:   for (int i=0; i<this->array_dimension_; i++)
209:                 if (this->elements_[i] != l.elements_[i])
210:                         return false;
211:         return true;
212: }


215: #endif // _LISTAVT_H
